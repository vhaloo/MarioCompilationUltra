cmake_minimum_required(VERSION 3.20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(MarioCompilationUltra VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 99)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
if(ANDROID)
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  add_compile_options("-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=mcu-src")
endif()
include(CTest)
include(FetchContent)
set(MCU_SDL_SOURCE "" CACHE PATH "Pinned SDL source for Android")
if(MCU_SDL_SOURCE)
  set(SDL_SHARED ON CACHE BOOL "" FORCE)
  set(SDL_STATIC OFF CACHE BOOL "" FORCE)
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  add_subdirectory("${MCU_SDL_SOURCE}" "${CMAKE_BINARY_DIR}/sdl2")
else()
  find_package(SDL2 2.28 CONFIG QUIET)
endif()
if(NOT TARGET SDL2::SDL2)
  set(SDL_SHARED ON CACHE BOOL "" FORCE)
  set(SDL_STATIC OFF CACHE BOOL "" FORCE)
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  set(SDL_ARCHIVE https://github.com/libsdl-org/SDL/releases/download/release-2.32.10/SDL2-2.32.10.tar.gz)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/SDL2-2.32.10.tar.gz")
    set(SDL_ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/SDL2-2.32.10.tar.gz")
  endif()
  FetchContent_Declare(sdl2 URL ${SDL_ARCHIVE} URL_HASH SHA256=5f5993c530f084535c65a6879e9b26ad441169b3e25d789d83287040a9ca5165)
  FetchContent_MakeAvailable(sdl2)
endif()
include(cmake/SnesCore.cmake)
add_library(mcu_zlib STATIC vendor/zlib/adler32.c vendor/zlib/compress.c vendor/zlib/crc32.c vendor/zlib/deflate.c vendor/zlib/infback.c vendor/zlib/inffast.c vendor/zlib/inflate.c vendor/zlib/inftrees.c vendor/zlib/trees.c vendor/zlib/uncompr.c vendor/zlib/zutil.c)
target_include_directories(mcu_zlib PUBLIC vendor/zlib)
add_library(mcu_imgui STATIC vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp vendor/imgui/backends/imgui_impl_sdl2.cpp vendor/imgui/backends/imgui_impl_sdlrenderer2.cpp)
target_include_directories(mcu_imgui PUBLIC vendor/imgui vendor/imgui/backends)
target_link_libraries(mcu_imgui PUBLIC SDL2::SDL2)
add_library(mcu_common STATIC src/files.cpp src/rom.cpp src/settings.cpp src/core.cpp src/world.cpp src/complete_map.cpp src/presentation.cpp src/localization.cpp src/spatial_controls.cpp src/object_previews.cpp src/mobile_controls.cpp src/android_bridge.cpp)
target_include_directories(mcu_common PUBLIC src vendor/snes9x/libretro)
target_link_libraries(mcu_common PUBLIC mcu_core mcu_zlib SDL2::SDL2)
if(ANDROID)
  option(MCU_BUILD_QUEST "Build the standalone OpenXR edition" ON)
  if(MCU_BUILD_QUEST)
  find_package(OpenXR REQUIRED CONFIG)
  add_library(MarioCompilationUltra SHARED src/main.cpp src/app.cpp src/ui_features.cpp src/mobile_ui.cpp src/quest.cpp src/quest_scene.cpp src/xr_stage.cpp)
  set_target_properties(MarioCompilationUltra PROPERTIES OUTPUT_NAME main)
  target_compile_definitions(MarioCompilationUltra PRIVATE MCU_QUEST XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_OPENGL_ES)
  target_link_libraries(MarioCompilationUltra PRIVATE OpenXR::openxr_loader android log EGL GLESv3)
  else()
    add_library(MarioCompilationUltra SHARED src/main.cpp src/app.cpp src/ui_features.cpp src/mobile_ui.cpp)
    set_target_properties(MarioCompilationUltra PROPERTIES OUTPUT_NAME main)
    target_compile_definitions(MarioCompilationUltra PRIVATE MCU_MOBILE)
    target_link_libraries(MarioCompilationUltra PRIVATE android log EGL GLESv2)
  endif()
  target_link_options(MarioCompilationUltra PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_executable(MarioCompilationUltra src/main.cpp src/app.cpp src/ui_features.cpp src/mobile_ui.cpp)
endif()
target_link_libraries(MarioCompilationUltra PRIVATE mcu_common mcu_imgui)
target_compile_definitions(MarioCompilationUltra PRIVATE SDL_MAIN_HANDLED MCU_VERSION="${PROJECT_VERSION}")
if(UNIX AND NOT APPLE)
  set_target_properties(MarioCompilationUltra PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
if(MSVC)
  foreach(t mcu_common MarioCompilationUltra mcu_imgui)
    target_compile_options(${t} PRIVATE /utf-8 /W3 /permissive-)
  endforeach()
  target_compile_definitions(mcu_core PRIVATE _CRT_SECURE_NO_WARNINGS __WIN32__)
  target_compile_options(mcu_core PRIVATE /utf-8 /wd4244 /wd4267 /wd4018 /wd4996)
  target_link_options(MarioCompilationUltra PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup)
endif()
if(WIN32)
  configure_file(assets/app.rc.in ${CMAKE_CURRENT_BINARY_DIR}/app.rc @ONLY)
  target_sources(MarioCompilationUltra PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
  target_link_libraries(MarioCompilationUltra PRIVATE shell32)
  add_custom_command(TARGET MarioCompilationUltra POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:MarioCompilationUltra>)
endif()
if(NOT ANDROID)
add_custom_command(TARGET MarioCompilationUltra POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:MarioCompilationUltra>/assets)
add_custom_command(TARGET MarioCompilationUltra POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/docs $<TARGET_FILE_DIR:MarioCompilationUltra>/docs)
endif()
install(TARGETS MarioCompilationUltra RUNTIME DESTINATION . LIBRARY DESTINATION .)
if(WIN32)
  install(FILES $<TARGET_FILE:SDL2::SDL2> DESTINATION .)
endif()
install(DIRECTORY assets DESTINATION .)
install(DIRECTORY docs DESTINATION .)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION . OPTIONAL)
install(DIRECTORY licenses DESTINATION .)
